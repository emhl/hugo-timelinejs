{{- $s := . -}} {{- $res := dict -}} {{/* Text */}} {{- $headline := $s.headline | default "" | markdownify -}} {{- $text := $s.text | default "" | markdownify -}} {{- if or $headline $text -}} {{- $res = merge $res (dict "text" (dict "headline" $headline "text" $text)) -}} {{- end -}} {{/* Dates */}} {{- with $s.date -}} {{- $res = merge $res (dict "start_date" (partial "timelinejs/parse-date.html" .)) -}} {{- end -}} {{- with $s.end -}} {{- $res = merge $res (dict "end_date" (partial "timelinejs/parse-date.html" .)) -}} {{- end -}} {{/* Media */}} {{- with $s.media -}} {{- if reflect.IsMap . -}} {{- $m := . -}} {{- if $m.caption -}} {{- $m = merge $m (dict "caption" ($m.caption | markdownify)) -}} {{- end -}} {{- if $m.credit -}} {{- $m = merge $m (dict "credit" ($m.credit | markdownify)) -}} {{- end -}} {{- $res = merge $res (dict "media" $m) -}} {{- else -}} {{- $res = merge $res (dict "media" (dict "url" .)) -}} {{- end -}} {{- end -}} {{/* Background */}} {{- with $s.background -}} {{- if reflect.IsMap . -}} {{- $res = merge $res (dict "background" .) -}} {{- else -}} {{- $res = merge $res (dict "background" (dict "url" .)) -}} {{- end -}} {{- end -}} {{/* Pass through other fields */}} {{- range $k, $v := $s -}} {{- if not (in (slice "headline" "text" "date" "end" "media" "background") $k) -}} {{- $val := $v -}} {{- if in (slice "display_date" "group") $k -}} {{- $val = $val | markdownify -}} {{- end -}} {{- $res = merge $res (dict $k $val) -}} {{- end -}} {{- end -}} {{- return $res -}}
